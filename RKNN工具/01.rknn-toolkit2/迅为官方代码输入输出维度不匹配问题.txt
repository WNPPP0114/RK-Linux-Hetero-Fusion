    # 1. 记录开始时间
    t_start = time.time()

    # --- 图像处理与推理 (保持你之前的逻辑) ---
    h_orig, w_orig = IMG.shape[:2]
    IMG_model = cv2.resize(IMG, (IMG_SIZE, IMG_SIZE))
    IMG_model = cv2.cvtColor(IMG_model, cv2.COLOR_BGR2RGB)
    
    outputs = rknn_lite.inference(inputs=[np.expand_dims(IMG_model, 0)])

    input0_data = outputs[0].reshape([3, -1] + list(outputs[0].shape[-2:]))
    input1_data = outputs[1].reshape([3, -1] + list(outputs[1].shape[-2:]))
    input2_data = outputs[2].reshape([3, -1] + list(outputs[2].shape[-2:]))

    input_data = list()
    input_data.append(np.transpose(input0_data, (2, 3, 0, 1)))
    input_data.append(np.transpose(input1_data, (2, 3, 0, 1)))
    input_data.append(np.transpose(input2_data, (2, 3, 0, 1)))

    boxes, classes, scores = yolov5_post_process(input_data)

    if boxes is not None:
        x_factor = w_orig / IMG_SIZE
        y_factor = h_orig / IMG_SIZE
        boxes[:, 0] *= x_factor
        boxes[:, 1] *= y_factor
        boxes[:, 2] *= x_factor
        boxes[:, 3] *= y_factor
        draw(IMG, boxes, scores, classes)
    # ----------------------------------------

    # 2. 计算延迟
    t_end = time.time()
    latency = (t_end - t_start) * 1000  # 转换为毫秒

    # 3. 把延迟写在视频左上角 (如果不想在视频上显示，可以注释掉这行)

                cv2.FONT_HERSHEY_SIMPLEX, 1, (0, 255, 0), 2)

    # 4. 【关键修改】同时返回图片和延迟数值
